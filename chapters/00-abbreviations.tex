\begin{abbreviations}
    \definition{БД}{база данных}
    \definition{ОЗУ}{оперативное запоминающее устройство}
    \definition{РБД}{реляционные базы данных}
    \definition{СУБД}{система управления базами данных}
    \definition{УДС}{улично-дорожная сеть}
    \definition{ЦП}{центральный процессор}
    \definition{3НФ}{третья нормальная форма реляционной модели данных}

    \definition{API}{Application Programming Interface (программный интерфейс приложения)}
    \definition{AoS}{Array of Structures (представление данных в виде массива структур)}
    \definition{CSR}{Compressed Sparse Row (сжатый формат хранения разреженных строк графа)}
    \definition{ELT}{Extract, Load, Transform (концепция извлечения, загрузки и преобразования геоданных)}
    \definition{EPSG}{European Petroleum Survey Group (стандарт идентификации геодезических систем координат)}
    \definition{GiST}{Generalized Search Tree (структура обобщенного поискового дерева пространственных индексов)}
    \definition{HFT}{High-Frequency Trading (класс архитектур сверхвысокой частоты обработки запросов)}
    \definition{IDEF0}{Integration Definition for Function Modeling (стандарт функционального моделирования систем)}
    \definition{IDEF1X}{Integration Definition for Information Modeling (стандарт информационного моделирования реляционных баз данных)}
    \definition{LCC}{Largest Connected Component (наибольшая сильно связанная компонента ориентированного графа)}
    \definition{MVT}{Mapbox Vector Tiles (бинарный картографический формат векторных тайлов на базе Protocol Buffers)}
    \definition{OSM}{OpenStreetMap (открытый проект совместного создания свободных географических карт)}
    \definition{POSIX}{Portable Operating System Interface (набор стандартов интерфейсов операционных систем)}
    \definition{RAM}{Random Access Memory (оперативная память с произвольным доступом)}
    \definition{REST}{Representational State Transfer (архитектурный стиль передачи состояния представлений)}
    \definition{SCC}{Strongly Connected Components (сильно связанные компоненты ориентированного графа)}
    \definition{SoA}{Structure of Arrays (представление данных в виде структуры массивов)}
    \definition{SQL}{Structured Query Language (декларативный язык структурированных запросов)}
    \definition{TTI}{Travel Time Index (индекс задержки времени поездки в транспортной сети)}
    \definition{WAL}{Write-Ahead Logging (протокол упреждающей записи в журнал транзакций СУБД)}
    \definition{WGS-84}{World Geodetic System 1984 (всемирная эллипсоидальная система географических координат)}
\end{abbreviations}

\begin{abbreviations}
    \definition{БД}{база данных}
    \definition{ЦП}{центральный процессор}

    \definition{API}{Application Programming Interface (программный интерфейс приложения)}
    \definition{BPR}{Bureau of Public Roads (функция задержки Бюро общественных дорог)}
    \definition{CSV}{Comma-Separated Values (текстовый формат представления табличных данных)}
    \definition{FIFO}{First-In-First-Out (обслуживание в порядке поступления)}
    \definition{GUI}{Graphical User Interface (графический интерфейс пользователя)}
    \definition{IPC}{Inter-Process Communication (межпроцессное взаимодействие)}
    \definition{JSON}{JavaScript Object Notation (текстовый формат обмена данными)}
    \definition{OSM}{OpenStreetMap (открытый картографический проект)}
    \definition{RPS}{Requests Per Second (число запросов, обрабатываемых за секунду)}
    \definition{SUMO}{Simulation of Urban MObility (пакет микро- и мезомоделирования транспортных потоков)}
    \definition{TTI}{Travel Time Index (индекс задержки времени поездки)}
    \definition{ZMQ}{ZeroMQ (библиотека асинхронного межпроцессного обмена сообщениями)}
\end{abbreviations}
